cmake_minimum_required(VERSION 3.20)
project(HyperFlow2D LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CHEM_PREFIX "$ENV{CONDA_PREFIX}" CACHE PATH "Cantera and SUNDIALS development prefix")
find_package(OpenMP REQUIRED)
find_path(CANTERA_INCLUDE cantera/base/Solution.h HINTS "${CHEM_PREFIX}/include" REQUIRED)
find_path(EIGEN_INCLUDE Eigen/Core HINTS "${CHEM_PREFIX}/include/eigen3" PATH_SUFFIXES eigen3 REQUIRED)
find_path(JSON_INCLUDE nlohmann/json.hpp HINTS "${CHEM_PREFIX}/include" REQUIRED)
set(CHEM_LIBRARIES)
foreach(lib cantera fmt sundials_cvode sundials_nvecserial sundials_sunmatrixdense sundials_sunlinsoldense sundials_core)
  find_library(LIB_${lib} NAMES ${lib} HINTS "${CHEM_PREFIX}/lib" REQUIRED)
  list(APPEND CHEM_LIBRARIES "${LIB_${lib}}")
endforeach()
include(FetchContent)
set(AMReX_SPACEDIM 2 CACHE STRING "" FORCE)
foreach(feature MPI FORTRAN AMRLEVEL PARTICLES LINEAR_SOLVERS EB INSTALL)
  set(AMReX_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(AMReX_OMP ON CACHE BOOL "" FORCE)
set(AMReX_ASSERTIONS ON CACHE BOOL "" FORCE)
FetchContent_Declare(amrex
  GIT_REPOSITORY https://github.com/AMReX-Codes/amrex.git
  GIT_TAG 2d8c2e7e9d782cee6a040c5a503497c4c0708c96
  GIT_SHALLOW FALSE)
FetchContent_MakeAvailable(amrex)
add_executable(hyperflow2d src/main.cpp src/physics.cpp src/chemistry.cpp
  src/chemistry_batch.cpp src/ConservativeInterp.cpp src/BlockAMR.cpp)
target_include_directories(hyperflow2d SYSTEM PRIVATE "${CANTERA_INCLUDE}" "${EIGEN_INCLUDE}" "${JSON_INCLUDE}")
target_link_libraries(hyperflow2d PRIVATE AMReX::amrex OpenMP::OpenMP_CXX ${CHEM_LIBRARIES})
target_compile_options(hyperflow2d PRIVATE -Wall -Wextra)
if(CHEM_PREFIX)
  set_target_properties(hyperflow2d PROPERTIES BUILD_RPATH "${CHEM_PREFIX}/lib")
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(boundary_check tests/boundary_check.cpp)
  target_include_directories(boundary_check PRIVATE src)
  target_include_directories(boundary_check SYSTEM PRIVATE "${CANTERA_INCLUDE}" "${EIGEN_INCLUDE}" "${JSON_INCLUDE}")
  target_link_libraries(boundary_check PRIVATE AMReX::amrex OpenMP::OpenMP_CXX)
  add_test(NAME boundary_check COMMAND boundary_check)
endif()
